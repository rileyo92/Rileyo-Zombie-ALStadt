#Treaty Ports
province_event = {
	id = 90954
	title = "EVTNAME90954"
	desc = "EVTDESC90954"

	trigger = {
		OR = {
			AND = { #Chinese Treaty Ports
				OR = {
					province_id = 1498 #Macau
					province_id = 1496 #Hong Kong
					province_id = 1608 
					province_id = 1606 
					province_id = 1538 #Shanghai
					province_id = 1569
					province_id = 1566 
					province_id = 1481 #Port Arther
				}
				NOT = { has_province_modifier = treaty_port }
				owner = {
					civilized = yes
					NOT = { primary_culture = nanfaren }
					NOT = { primary_culture = manchu }
				}
			}
			AND = { #Indian Treaty Ports
				OR = {
					province_id = 1303 #Goa
					province_id = 1312 #Pondicherry
				}
				NOT = { has_province_modifier = treaty_port }
				owner = {
					civilized = yes
				}
			}
			AND = { #Malayan Treaty Ports
				OR = {
					province_id = 1388 #
					province_id = 1386 #
					province_id = 1384 #Singapore
				}
				NOT = { has_province_modifier = treaty_port }
				owner = {
					civilized = yes
				}
			}
			AND = { #Suez Canal
				OR = {
					province_id = 1755 #Suez Canal
				}
				NOT = { has_province_modifier = treaty_port }
				owner = {
					civilized = yes
				}
			}
		}
	}

	mean_time_to_happen = { months = 1 }

	option = {
		name = "EVTOPTA90954"
		trade_goods = precious_metal
		any_pop = { reduce_pop = 1.2 }
		add_province_modifier = { name = treaty_port duration = -1 }
	}
}

	title = "EVTNAME90954"
	desc = "EVTDESC90955"

		}
	}

	mean_time_to_happen = { months = 1 }

		}
	}
}
